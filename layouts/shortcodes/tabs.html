{{- /* The tab markup lives in layouts/partials/tabs.html, which the code_sample shortcode also calls. This shortcode resolves each tab's content, which needs the page, and hands the rendered result over. */ -}} {{- .Page.Scratch.Add "tabset-counter" 1 -}} {{- $tabPaneOrdinal := .Get "name" | default (printf "tabset-%s-%d" (.Page.RelPermalink) (.Page.Scratch.Get "tabset-counter") ) | anchorize -}} {{- if .Inner -}}{{- /* We don't use the inner content, but Hugo will complain if we don't reference it. */ -}}{{- end -}} {{- $tabs := slice -}} {{- range $e := .Scratch.Get "tabs" -}} {{- $content := "" -}} {{- with index $e "content" -}} {{- $content = . -}} {{- else -}} {{- if eq $.Page.BundleType "leaf" -}} {{- /* find the file somewhere inside the bundle. Note the use of double asterisk */ -}} {{- with $.Page.Resources.GetMatch (printf "**%s*" $e.include) -}} {{- if ne .ResourceType "page" -}} {{- /* Assume it is a file that needs code highlighting. */ -}} {{- $codelang := $e.codelang | default ( path.Ext .Name | strings.TrimPrefix ".") -}} {{- $content = highlight .Content $codelang "" -}} {{- else -}} {{- $content = .Content -}} {{- end -}} {{- end -}} {{- else -}} {{- $path := path.Join $.Page.File.Dir $e.include -}} {{- with site.GetPage "page" $path -}} {{- $content = .Content -}} {{- else -}} {{- errorf "[%s] tabs include not found for path %q" site.Language.Lang $path -}} {{- end -}} {{- end -}} {{- end -}} {{- $tabs = $tabs | append (dict "label" (index $e "name" | markdownify) "content" $content "persist" "" "disabled" (index $e "disabled" | default false)) -}} {{- end -}} {{- /* tabindex is "0", not the tab set's slug: the attribute takes a number, and a pane is the kind of region a keyboard has to be able to reach. */ -}} {{- partial "tabs.html" (dict "id" $tabPaneOrdinal "ariaLabel" "" "paneClass" "tab-body" "tabindex" "0" "tabs" $tabs) -}}
